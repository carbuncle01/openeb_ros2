cmake_minimum_required(VERSION 3.24)
# Independent entry point: no ament, ROS, OpenEB SDK, TensorRT or camera.
project(openeb_cuda_benchmarks LANGUAGES CXX)
option(OPENEB_BENCH_CPU_ONLY "Build portable correctness tests without CUDA" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(OPENEB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
enable_testing()
foreach(name evt3_runs_histogram_test histogram_bin_updates_test)
  add_executable(${name} "${OPENEB_ROOT}/test/${name}.cpp")
  target_include_directories(${name} PRIVATE "${OPENEB_ROOT}/include")
  add_test(NAME ${name} COMMAND ${name})
endforeach()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME notebook_sweep_validation COMMAND "${Python3_EXECUTABLE}" "${OPENEB_ROOT}/test/test_cuda_notebook_tools.py")
endif()
option(OPENEB_BENCH_RAW "Build RAW replay using external event_camera_codecs headers" OFF)
if(OPENEB_BENCH_RAW)
  find_path(EVENT_CAMERA_CODECS_INCLUDE_DIR event_camera_codecs/evt3_decoder.h REQUIRED)
  file(SHA256 "${EVENT_CAMERA_CODECS_INCLUDE_DIR}/event_camera_codecs/evt3_decoder.h" CODEC_SHA256)
  add_executable(raw_evt3_benchmark_input_test "${OPENEB_ROOT}/test/raw_evt3_benchmark_input_test.cpp")
  target_include_directories(raw_evt3_benchmark_input_test PRIVATE "${OPENEB_ROOT}/include" "${EVENT_CAMERA_CODECS_INCLUDE_DIR}")
  add_test(NAME raw_evt3_benchmark_input_test COMMAND raw_evt3_benchmark_input_test)
endif()
if(NOT OPENEB_BENCH_CPU_ONLY)
  # Override with a numerical SM if cross-compiling or no GPU is visible.
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(openeb_cuda_backend STATIC "${OPENEB_ROOT}/src/event_tensor_cuda_backend.cu")
  target_include_directories(openeb_cuda_backend PUBLIC "${OPENEB_ROOT}/include")
  target_link_libraries(openeb_cuda_backend PUBLIC CUDA::cudart)
  set_target_properties(openeb_cuda_backend PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  foreach(name histogram_bin_cuda_benchmark event_tensor_cuda_load_benchmark histogram_bin_cuda_test event_tensor_cuda_backend_test)
    add_executable(${name} "${OPENEB_ROOT}/test/${name}.cpp")
    target_link_libraries(${name} PRIVATE openeb_cuda_backend)
  endforeach()
  if(OPENEB_BENCH_RAW)
    add_executable(event_tensor_raw_cuda_benchmark "${OPENEB_ROOT}/test/event_tensor_raw_cuda_benchmark.cpp")
    target_link_libraries(event_tensor_raw_cuda_benchmark PRIVATE openeb_cuda_backend)
    target_include_directories(event_tensor_raw_cuda_benchmark PRIVATE "${EVENT_CAMERA_CODECS_INCLUDE_DIR}")
    target_compile_definitions(event_tensor_raw_cuda_benchmark PRIVATE OPENEB_CODEC_DECODER_SHA256="${CODEC_SHA256}")
  endif()
  add_test(NAME histogram_bin_cuda_test COMMAND histogram_bin_cuda_test)
  add_test(NAME event_tensor_cuda_backend_test COMMAND event_tensor_cuda_backend_test)
endif()
